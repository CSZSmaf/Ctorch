cmake_minimum_required(VERSION 3.24)
project(CtorchConsumer LANGUAGES C CXX)
find_package(Ctorch 0.1 CONFIG REQUIRED)
enable_testing()
add_executable(cpp_consumer main.cpp)
target_link_libraries(cpp_consumer PRIVATE ctorch::ctorch)
add_test(NAME cpp_consumer COMMAND cpp_consumer)
add_executable(c_consumer ../c_api.c)
target_compile_features(c_consumer PRIVATE c_std_17)
target_link_libraries(c_consumer PRIVATE ctorch::ctorch)
set_target_properties(c_consumer PROPERTIES C_EXTENSIONS OFF LINKER_LANGUAGE CXX)
add_test(NAME c_consumer COMMAND c_consumer)
if(WIN32 AND Ctorch_SHARED)
    foreach(target IN ITEMS cpp_consumer c_consumer)
        add_custom_command(TARGET ${target} POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different
                $<TARGET_FILE:ctorch::ctorch> $<TARGET_FILE_DIR:${target}>)
    endforeach()
endif()
